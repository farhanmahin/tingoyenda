const selectedBooks = new Set();

function renderBooks() {
  const grid = document.getElementById('booksGrid');
  grid.innerHTML = '';
  
  bookList.forEach((book, index) => {
    const btn = document.createElement('div');
    btn.className = 'tg-book-btn';
    btn.innerHTML = `<span>📖</span> <span>${index + 1}. ${book}</span>`;
    btn.onclick = () => toggleBook(index, btn);
    grid.appendChild(btn);
  });
}

function toggleBook(index, element) {
  if (selectedBooks.has(index)) {
    selectedBooks.delete(index);
    element.classList.remove('selected');
  } else {
    selectedBooks.add(index);
    element.classList.add('selected');
  }
  updateProgress();
}

function updateProgress() {
  const count = selectedBooks.size;
  const percentage = Math.round((count / bookList.length) * 100);
  
  document.getElementById('selectedCount').innerText = count;
  document.getElementById('percentageText').innerText = percentage + '%';
  document.getElementById('progressBar').style.width = percentage + '%';
}

function toBengaliNumber(n) {
  const bnNums = ['০', '১', '২', '৩', '৪', '৫', '৬', '৭', '৮', '৯'];
  return n.toString().replace(/\d/g, d => bnNums[d]);
}

function getNumericBengaliDate() {
  const now = new Date();
  const day = String(now.getDate()).padStart(2, '0');
  const month = String(now.getMonth() + 1).padStart(2, '0');
  const year = now.getFullYear();
  
  const formattedStr = `${day}/${month}/${year}`;
  return formattedStr.replace(/\d/g, d => ['০', '১', '২', '৩', '৪', '৫', '৬', '৭', '৮', '৯'][d]);
}

function generatePhotoCard() {
  const nameInput = document.getElementById('readerName').value.trim();
  
  if (nameInput === "") {
    alert("দয়া করে আপনার নামটি লিখুন!");
    document.getElementById('readerName').focus();
    return;
  }
  
  const count = selectedBooks.size;
  const percentage = Math.round((count / bookList.length) * 100);

  document.getElementById('cardTitleText').innerText = nameInput;
  document.getElementById('cardCountText').innerText = toBengaliNumber(count);
  document.getElementById('cardProgressFill').style.width = `${percentage}%`;
  document.getElementById('cardPercentText').innerText = `${toBengaliNumber(percentage)}%`;
  document.getElementById('cardDateText').innerText = getNumericBengaliDate();

  const outputSec = document.getElementById('card-output-section');
  outputSec.style.display = 'block';
  outputSec.scrollIntoView({ behavior: 'smooth' });
}

/* Enhanced High-Quality Photo Download Function */
function downloadPhotoCard() {
  const cardElement = document.getElementById('photo-card');

  html2canvas(cardElement, {
    scale: 4, // Higher scale for crystal-clear HD resolution
    useCORS: true,
    backgroundColor: '#fffef9',
    logging: false
  }).then(canvas => {
    const link = document.createElement('a');
    link.download = 'tin-goyenda-photo-card.png';
    link.href = canvas.toDataURL('image/png', 1.0);
    link.click();
  });
}
